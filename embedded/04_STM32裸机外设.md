```c
 * EMBEDDED INTERVIEW PREP

 * 文件：04_Bare_Metal_Peripherals/01_gpio_registers.c

#include <stdio.h>
#include <stdint.h>
#include <assert.h>

/* ===============================================================
 * THEORY — STM32 GPIO 寄存器映射
 * ===============================================================
 *
 * 每个 GPIO 端口有 10 个寄存器（10×4 字节 = 40 字节）。
 *
 * MODER (0x00)：模式 — 每个引脚 2 位
 * 00 = 输入，01 = 输出，10 = 复用功能，11 = 模拟
 *
 * OTYPER (0x04)：输出类型 — 每个引脚 1 位
 * 0 = 推挽式，1 = 漏极开路
 *
 * OSPEEDR (0x08)：输出速度 — 每个引脚 2 位
 * 00=低，01=中，10=高，11=非常高
 *
 * PUPDR (0x0C)：上拉/下拉 — 每个引脚 2 位
 * 00=无，01=上拉，10=下拉
 *
 * IDR (0x10)：输入数据寄存器（只读，每个引脚 1 位）
 * ODR (0x14)：输出数据寄存器（每个引脚 1 位）
 *
 * BSRR (0x18)：位设置/复位寄存器 — ATOMIC 操作
 * 位 [15:0] = SET（写入 1 到设置引脚，0 = 无效）
 * 位 [31:16] = RESET（向清除引脚写入 1，0 = 无效）
 *
 * AFRL (0x20)：复用功能低电平（引脚 0-7，每个 4 位）
 * AFRH (0x24)：复用功能高位（引脚 8-15，每个 4 位）
 *
 * 时钟使能：RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN；
 * MUST 可以通过 BEFORE 访问任何 GPIO 寄存器。
 * =============================================================== */


/* 模拟GPIO和RCC寄存器进行主机端练习 */
typedef struct {
    volatile uint32_t MODER;
    volatile uint32_t OTYPER;
    volatile uint32_t OSPEEDR;
    volatile uint32_t PUPDR;
    volatile uint32_t IDR;
    volatile uint32_t ODR;
    volatile uint32_t BSRR;
    volatile uint32_t LCKR;
    volatile uint32_t AFRL;
    volatile uint32_t AFRH;
} GPIO_TypeDef;

typedef struct {
    volatile uint32_t AHB1ENR;
    volatile uint32_t APB1ENR;
    volatile uint32_t APB2ENR;
} RCC_TypeDef;

/* 位定义 */
#define RCC_AHB1ENR_GPIOAEN  (1u << 0)
#define RCC_AHB1ENR_GPIOBEN  (1u << 1)
#define RCC_AHB1ENR_GPIOCEN  (1u << 2)

/* GPIO 模式常数 */
#define GPIO_MODE_INPUT    0b00u
#define GPIO_MODE_OUTPUT   0b01u
#define GPIO_MODE_AF       0b10u
#define GPIO_MODE_ANALOG   0b11u

/* GPIO 速度常数 */
#define GPIO_SPEED_LOW     0b00u
#define GPIO_SPEED_MEDIUM  0b01u
#define GPIO_SPEED_HIGH    0b10u
#define GPIO_SPEED_VHIGH   0b11u

/* GPIO 拉力常数 */
#define GPIO_PULL_NONE     0b00u
#define GPIO_PULL_UP       0b01u
```

```c
/* ================================================================ TASK 1 */

void gpio_clock_enable(RCC_TypeDef *rcc, uint8_t port_index)
{
    rcc->AHB1ENR |= (1u << port_index);
}

void gpio_set_mode(GPIO_TypeDef *gpio, uint8_t pin, uint8_t mode)
{
    gpio->MODER = (gpio->MODER & ~(0b11u << (pin*2))) | ((uint32_t)mode << (pin*2));
}

void gpio_set_speed(GPIO_TypeDef *gpio, uint8_t pin, uint8_t speed)
{
    gpio->OSPEEDR = (gpio->OSPEEDR & ~(0b11u << (pin*2))) | ((uint32_t)speed << (pin*2));
}

void gpio_set_pull(GPIO_TypeDef *gpio, uint8_t pin, uint8_t pull)
{
    gpio->PUPDR = (gpio->PUPDR & ~(0b11u << (pin*2))) | ((uint32_t)pull << (pin*2));
}

void gpio_set_output_type(GPIO_TypeDef *gpio, uint8_t pin, uint8_t otype)
{
    if (otype) gpio->OTYPER |=  (1u << pin);
    else       gpio->OTYPER &= ~(1u << pin);
```

```c
void gpio_write_pin(GPIO_TypeDef *gpio, uint8_t pin, uint8_t value)
{
```

```c
/* ================================================================ TASK 3 */

void gpio_set_af(GPIO_TypeDef *gpio, uint8_t pin, uint8_t af_num)
{
    if (pin < 8) {
        uint8_t shift = pin * 4u;
        gpio->AFRL = (gpio->AFRL & ~(0xFu << shift)) | ((uint32_t)af_num << shift);
    } else {
        uint8_t shift = (pin - 8u) * 4u;
        gpio->AFRH = (gpio->AFRH & ~(0xFu << shift)) | ((uint32_t)af_num << shift);
    }
```

### 对应参考答案

```c
/* ================================================================ TASK 6 — 错误搜寻 FIXED

Bug 1：在访问 GPIO 寄存器之前，RCC 时钟未启用。
   在 STM32 上，默认情况下禁用外设时钟以节省功耗。
   写入 GPIO 寄存器而不启用其时钟没有任何效果
   （或导致某些设备上出现总线故障）。
   FIX: RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;在任何 GPIO 访问之前。

错误 2：gpio->MODER |= (GPIO_MODE_AF << (引脚*2))
   这会将新模式与先清除的现有位 WITHOUT 进行或运算。
   如果引脚处于 AF 模式 (0b10) 并且您在 0b10 中执行 OR → 没有变化，则正确。
   但如果引脚处于 OUTPUT 模式 (0b01) 并且您尝试设置 INPUT (0b00)：
   0b01 | 0b01 0b00 = 0b01 → 仍然输出！明确的步骤是强制性的。
   FIX: GPIO->MODER = (GPIO->MODER & ~(0b11u<<(引脚*2))) | (模式<<(引脚*2));

Bug 3：gpio->OTYPER &= ~(1u << pin) 设置 PUSH-PULL（位 = 0）。
   I2C SDA 必须是 OPEN-DRAIN（位 = 1） — 多个设备共享线路。
   推挽模式下，如果一台设备驱动HIGH，另一台设备驱动LOW
   同时→短路→硬件损坏。
   FIX: GPIO->OTYPER |= (1u << 引脚);  // 设置漏极开路*/

/*===============================================================
 * INTERVIEW QUESTION ANSWERS
 * ===============================================================

Q1：为什么使用BSRR而不是ODR？ BSRR 消除了什么竞争条件？

A：ODR 需要读-修改-写：读 ODR → 修改位 → 写 ODR。
   在读取和写入之间，ISR 可以触发并修改 ODR
   对于不同的引脚。当main的写入完成时，它会覆盖ISR的
   更改 → ISR 设置的引脚被清除。
   BSRR：单个 32 位写入原子设置或重置一个或多个引脚
   不影响任何其他引脚。无需读取 → 无竞争条件。
   位 [15:0] = 设置掩码，位 [31:16] = 重置掩码。一次写入=原子。

Q2：PA5 配置为 OUTPUT，但读取 IDR.5 始终返回 0。

答：可能的原因：
   (a) 外部硬件：有东西将引脚拉至 GND（短路）。
       如果输出正在驱动 HIGH，但 IDR 读取为 0，则该引脚被过驱动。
   (b) MODER 配置不正确 — 仍处于 INPUT 或 ANALOG 模式。
       检查 MODER[11:10] — 输出应为 0b01。
   (c) 时钟未启用 — GPIO 外设未提供时钟，寄存器写入被忽略。
   (d) ODR 位未设置 — MODER=输出，但 ODR.5=0 → 引脚为 LOW → IDR.5=0（正确！）。
   注：在输出模式下，IDR 反映的是实际引脚电压，而不是 ODR。
   即使 ODR.5=1，外部拉动 LOW 的输出引脚也会显示 IDR.5=0。

Q3：如果忘记启用 GPIO 外设时钟会怎样？

答：在 STM32 上，所有外设时钟默认都是门控的（低功耗）。
   写入 GPIO 寄存器而不启用时钟：
   - APB/AHB 总线写入可能会无错误地完成（无总线故障）。
   - 但写入被丢弃 - 外设没有时钟、寄存器
     不要锁存这些值。
   - GPIO 引脚保持默认状态（输入，无上拉）。
   无声的失败——初学者很常见的错误。始终首先启用时钟。

Q4：来自 3.3V MCU 的 5V I2C 总线上的开漏输出。

A：配置为开漏输出（OTYPER 位 = 1）。
   在开漏模式下：MCU 只能拉线 LOW（驱动 0）。
   要转到 HIGH，MCU 释放线路（不驱动任何东西），然后
   外部上拉电阻拉至 5V。
   MCU 的输入保护二极管可安全处理 5V HIGH 电平
   （只要引脚能够承受 5V 电压——请检查数据表！）。
   如果配置为推挽式并在 5V 总线上以 3.3V 驱动 HIGH：
   - 与驱动 LOW 的其他 5V 设备短路。
   - 3.3V HIGH 可能不符合 5V I2C VOH 规格 (0.7×VCC = 3.5V)。

Q5：GPIO 可以与 BSRR 同时设置多少个引脚？原子？

答：一个端口的所有 16 个引脚同时进行一次 32 位写入。

   是的，这是原子性的：单个 32 位 AHB 总线写入在 Cortex-M 上是原子性的。
   GPIO 外设在同一时钟周期锁存所有位。*/
```

```c
#define TX_BUF_MASK  (TX_BUF_SIZE - 1)

typedef struct {
    uint8_t  buf[TX_BUF_SIZE];
    volatile uint8_t head, tail;
} TxRingBuf;

{

        usart->CR1 &= ~USART_CR1_TXEIE;   /* 没有更多数据——禁用中断 */
```

```c
   TC（传输完成，SR 位 6）：当 LAST 位移位时触发
   TX 线上的 AND TXE 已设置。电线闲置。
   将 TC 用于 RS-485 DE（方向使能）引脚：您必须将 NOT 置低 DE，直到
   TC 会触发——否则你会在传输过程中切断最后的比特。

Q3：为什么在发送功能中启用TXEIE，而在完成后在ISR中禁用？

答：只要 TXE=1，TXEIE（TX 空中断使能）就会连续触发
   （DR 为空）。如果我们永久启用它，ISR 将发射数千次
   即使没有任何内容可发送，每秒也会发送多次 — 浪费 CPU。
   图案：
   - uart_send()：推入环形缓冲区，然后|= TXEIE。 ISR 立即开火。
   - ISR：如果环形缓冲区有数据→写入DR。如果为空→清除TXEIE。完毕。
   这是“具有自动禁用功能的中断驱动 TX”模式。高效的。

Q4：RS-485 带 UART。 GPIO 引脚和时序。

答：RS-485 需要一个 DE（驱动器使能）引脚来在发送和接收之间切换
   （半双工总线）。将 MCU GPIO 连接到 RS-485 收发器的 DE/RE# 引脚。
   发射前：GPIO HIGH（启用发射器，禁用接收器）。
   发送后：等待TC（发送完成）标志，然后GPIO LOW
   （禁用发射器，启用接收器）。
   MUST 使用 TC 标志，而不是 TXE — 当 DR 为空但最后一个字节为 TXE 时触发
   仍在移位寄存器中。过早切换 DE 会切断最后的位。

Q5：115200 波特时的位时间和帧持续时间。 ISR 延迟约束。

答：位时间 = 1 / 115200 = 8.68 µs。
   8N1 帧 = 1 个起始 + 8 个数据 + 1 个停止 = 10 位 = 每字节 86.8 µs。
   在 115200 波特率下，每 86.8 µs 就会有一个新字节到达。
   ISR 延迟约束：ISR 必须先读取 DR 寄存器（清零 RXNE）
   NEXT 字节开始到达并溢出 1 字节硬件缓冲区。
   最大 ISR 延迟：86.8 µs（一帧时间）。
   在 168 MHz 时：86.8 µs = ~14,600 CPU 周期 — 非常慷慨。即使是慢速的 ISR 也可以。
   在 921600 波特率下：帧 = 10.8 µs → ~1,815 个周期。 ISR一定要快。
*/
```

```c
/*===============================================================
 * INTERVIEW QUESTION ANSWERS（代码前填写）
 * ================================================================

Q1：8 个传感器 — SPI 与 I2C？

答：取决于传感器。
   SPI：8 个传感器需要 8 个 CS 引脚 + 3 个共享引脚 (SCLK/MOSI/MISO) = 总共 11 个引脚。
   优点：全双工、快速（10-50 Mbit/s）、无地址冲突。
   缺点：管脚较多，走线较长。
   I2C：8个传感器共用2根线（SDA+SCL）。需要 8 个唯一地址。
   优点：只有2个引脚。缺点：半双工，最大400kHz（快速）
   或 1MHz (Fast+)，地址空间有限（127 个地址，部分保留）。
   如果满足以下条件，请选择 SPI：高数据速率、需要抗噪性（差分）、长时间运行 PCB。
   如果满足以下条件，请选择 I2C：低速传感器（温度、湿度、RTC）、引脚数至关重要。

Q2：I2C 总线卡住 — SCL 低，SDA 低。原因及恢复。

A：原因：断电或MCU复位时从机处于中字节。奴隶
   保持 SDA 为低电平，等待更多时钟脉冲来完成其字节。
   恢复过程（来自 GPIO 的位爆炸）：
   1. 检查 SCL — 如果卡在低电平，则存在硬件问题（短路）。
      如果 SDA 卡住并且 SCL 空闲：
   2. 在 SCL 上发送最多 9 个时钟脉冲（GPIO 切换）。
      大多数从机在看到其字节的剩余时钟后将释放 SDA。
   3. 生成 STOP 条件（SDA 低 → SDA 高，同时 SCL 高）。
   4. 如果仍然卡住：重新启动从站。
   Linux：I2C 子系统中的 `i2c_recover_bus()` 自动执行此操作。
   在 STM32 上：使用 I2C 外设复位（CR1.SWRST），然后 GPIO 恢复。

Q3：I2C 中的时钟拉伸。哪个设备执行它？

答：SLAVE 保持 SCL 为低电平以暂停主时钟。
   当主机将 SCL 驱动为高电平时，从机将其保持为低电平（漏极开路总线）。
   主设备的 SCL 输入看到 LOW，等待直到看到 HIGH，然后继续。
   使用案例：从机收到 I2C 地址 + 寄存器，需要时间来获取数据
   在发送响应之前，先从内部 EEPROM 或 ADC 开始。
   没有时钟拉伸：主机将在从机时输入垃圾数据
   还在准备中。
   主控要求：SCL 必须是漏极开路（或具有时钟拉伸）
   检测电路）——推挽式 SCL 不能被从机保持为低电平。

Q4：CPOL=1，CPHA=1 字节 0b10110001 的波形。

A：模式3：时钟空闲HIGH（CPOL=1），数据在上升沿采样（CPHA=1）。
   0b10110001 的序列（首先是 MSB）：
   SCLK： H H L H L H L H L H L H L H L H H
          |下降|上升|下降|上升| ...
   DATA：\_1___0___1___1___0___0___0___1/
   - SCLK 下降沿数据变化。
   - 在 SCLK 的上升沿采样（读取）数据。
   - CS 在第一个时钟之前置为低电平，在第 8 个上升沿后置低。

Q5：I2C 上拉电阻 — 太大还是太小。

答：I2C 上的上拉电阻为 MANDATORY（开漏线无法主动驱动 HIGH）。
   太大（例如 100 kΩ）：
   - RC 时间常数 = R × C_bus 太长。信号缓慢上升。
   - 在 400 kHz 时，在采样之前位可能无法到达 VIH → 帧错误。
   - 标准规定快速模式的上升时间 < 300 ns。
   太小（例如 100 Ω）：
   - 当从机拉动 SDA LOW 时，每线电流 = VCC/R = 3.3V/100Ω = 33 mA。
   - 超过从设备的输出灌电流规格（通常为 3-10 mA）→ VOL 过高。
   - 还浪费电力。
   正确值：100 kHz 为 4.7 kΩ，400 kHz 为 2.2 kΩ（3.3V、<100pF 总线的典型值）。
   公式：R = (VCC - VOL_max) / I_OL_min，受上升时间约束 = 0.8473 × R × C_bus。

Q6：I2C 7 位寻址与 10 位寻址。

A：7位（标准）：地址为7位→128个地址，部分保留→~112个可用。
   帧：START + [ADDR:7][R/W:1]（1 字节）+ ACK + ...
   10位：地址为10位→1024个地址。
   帧：START + [11110:5][ADDR_HIGH:2][W:1]（字节1）+ ACK +
```

```c
/* ===============================================================
 * THEORY — STM32 通用定时器 (TIM2–TIM5)
 * ===============================================================
 *
 * 定时器每 N 个时钟周期递增其计数器 CNT，
 * 其中 N 由预分频器 (PSC) 设置。
 *
 * 定时器时钟频率：FCLK（经过APB预分频器）
 * 计数器时钟：FCLK / (PSC + 1)
 *更新事件（UEV）：当CNT达到ARR时发生（自动重新加载）
 *
 * 公式：
 * 更新频率 = FCLK / ((PSC+1) * (ARR+1))
 * 周期 (s) = 1 / Update_Frequency
 *
 * 示例：84 MHz 定时器时钟的 1 ms 刻度
 * 84,000,000 / ((839+1) * (99+1)) = 1000 Hz → 1 ms 周期
 * PSC = 839，ARR = 99
 *
 * 关键寄存器：
 * CR1 : CEN (bit0) — 计数器使能，DIR (bit4) — 向上/向下
 * PSC ：预分频器值（PSC+1 对输入时钟进行分频）
 * ARR ：自动重载寄存器（计数器在 ARR 处回绕）
 * CNT : 当前计数器值
 * CCR1-4：捕捉/比较寄存器（用于PWM或输入捕捉）
 * CCMR1 ：通道 1/2 模式（OC1M 位 [6:4] 用于 PWM 模式）
 * CCER : 捕捉/比较使能
 * DIER：中断/DMA 使能（UIE bit0 = 更新中断）
 * SR：状态（UIF bit0 = 更新中断标志）
 * EGR ：事件生成（UG bit0 = 强制更新）
 *
 * PWM 模式 1：CNT < CCRx → 输出 HIGH
 * CNT >= CCRx → 输出 LOW
 * 占空比 = CCRx / (ARR+1) × 100%
 * =============================================================== */
```

```c
Q2：PWM 占空比停留在 0%。调试清单。

答： 1. 检查 CCER：CC1E 位（通道 1 输出使能）必须置位 (=1)。
      如果CC1E=0，则该引脚始终由GPIO默认驱动，而不是PWM。
   2. 检查CCR1值：如果PWM模式1下CCR1=0（CNT<CCR→HIGH），
      0 计数的输出为 HIGH → 有效 0% 占空比。
   3. 检查CCMR1：OC1M[2:0]必须为110（PWM 模式1）或111（模式2）。
      如果OC1M=000（冻结），输出不会改变。
   4. 检查 GPIO 复用功能：引脚必须处于 AF 模式且 AF# 正确。
      如果引脚处于 GPIO 输出模式，则该引脚上不会出现 PWM 信号。
   5. 检查 ARR 预载：如果更改 CCR，则应设置 OC1PE（CCMR1 位 3）
      操作期间——防止出现故障。
   6. 检查BDTR（高级定时器TIM1/TIM8）：MOE（主输出使能）位
      必须设置。高级定时器需要 MOE=1 才能在引脚上显示输出。

Q3：PWM 频率和占空比测量的输入捕获。

答：在同一引脚上使用两个通道 (CC1S/CC2S)：
   CC1：上升沿触发（IC1F=00，CC1P=0=上升沿）
   CC2：同一引脚上下降沿触发（CC2S=10=IC2映射到TI1，
        CC2P=1=下降）
   周期测量：上升沿捕获CC1→等待→再次捕获CC1。
   频率 = TIMER_CLOCK / (CC1[2] - CC1[1]) -- 或 CC1 的使用周期
   占空比：捕获CC2（下降沿）和CC1（下一个上升沿）。
   high_time = CC2 - CC1（前一个上升）
   周期 = CC1_新 - CC1_旧
   占空比 = (high_time * 100) / 周期
   复位 CC1 捕获上的计数器（TI1FP1 上的 SMS=100 复位模式）以进行干净测量。

答：FreeRTOS 刻度：SysTick 每 1 毫秒（或 1/configTICK_RATE_HZ 秒）触发一次。

   配置MAX_SYSCALL_INTERRUPT_PRIORITY）。

   没有 RTOS 参与。阻塞延迟。

答：PSC=0：定时器每 1/168MHz = ~5.95 ns 递增。
   这是定时器分辨率。
```
