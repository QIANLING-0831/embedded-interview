/* ── 任务4──────────────────────────────────────────────────── */

uint8_t count_set_bits(uint32_t x)
{
    uint8_t count = 0;
    while (x) { x &= x - 1; count++; }   /* 布赖恩·科尼汉 */
    return count;
}

int8_t find_highest_set_bit(uint32_t x)
{
    if (x == 0) return -1;
    int8_t pos = 0;
    while (x >>= 1) pos++;
    return pos;
}

uint32_t reverse_bits(uint32_t x)
{
    uint32_t result = 0;
    for (int i = 0; i < 32; i++) {
        result = (result << 1) | (x & 1u);
        x >>= 1;
    }
    return result;
}

/* ── 任务 6 — Bug 搜寻答案 ──────────────────────────────── */
/*
 * Bug 1: reg = reg & mask;
 * 这 KEEPS 只有字段位 — 应该是 CLEAR。
 * FIX: reg = reg & ~mask;   （面具的补充）
 *
 * 错误 2：uint16_t 字段 = val << 移位；
 * val 是 uint8_t（8 位）。对于 8 位类型，左移 12 就是 UB。
 * FIX: uint16_t 字段 = (uint16_t)((uint16_t)val << 移位);
 *（首先将 val 转换为 uint16_t，THEN 移位）
 *
 * 错误 3：返回 reg |场地;
 * 此时 reg 仍然具有旧的字段位，因为 Bug 1 是错误的。
 *修复Bug 1后，reg被正确屏蔽，这个OR是正确的。
 * BUT：掩码补码逻辑必须位于 OR 之前。
 *
 * 更正功能：
 * uint16_t set_field（uint16_t reg，uint8_t val）{
 * uint16_t 掩码 = 0xF000;
 * uint8_t 移位 = 12;
 * reg &= ~掩码;                              // 清除字段
 * 返回寄存器| ((uint16_t)((uint16_t)val << 移位) & 掩码);
 * }
 */

/* ── 面试问题解答──────────────────────────────── */
/*
 * Q1: (1<<31) 是有符号整型移位——大多数平台上的 UB。
 * (1u<<31) 是无符号的 — 定义的行为，给出 0x80000000。
 *
 * Q2：在 Cortex-M 上，使用 LDREX/STREX（独占访问）或禁用中断
 * __disable_irq()/__enable_irq() 围绕读-修改-写。
 * 许多STM32外设也有BSRR寄存器：原子设置/清除
 * 一次写入 — 无需读取-修改-写入。
 *
 * Q3: reg &= ~0x0Fu;   （清除位 3:0，保留所有其他位）
 *
 * Q4：位带将每个位映射到字对齐地址，从而形成 32 位
 * 写入 = 单比特切换。消除了读-修改-写，所以它是
 * 从 ISR 的角度来看，本质上是原子的。在 Cortex-M7+ 上已弃用。
 *
 * Q5：在编译时使用 a 预计算 256 个条目（每个字节值一个）
 * constexpr/const 表，然后每个输入字节的 XOR 表[字节]。
 * ROM 总计：对于 CRC-16，256 × 2 字节 = 512 字节。*/

#include <stdint.h>

/* ===============================================================
 * THEORY — 结构填充和对齐
 * ===============================================================
 *
 * 编译器插入填充以使成员与其大小对齐：
 * 结构体 { uint8_t a; uint32_t b； } → sizeof = 8（a 之后填充 3 个字节）

typedef struct __attribute__((packed)) {
    uint8_t  sof;       /* 0xA5 */
    uint8_t  cmd;
    uint16_t value;     /* 小尾数法 */
    uint8_t  checksum;  /* XOR 的 cmd ^ value_lo ^ value_hi */

void frame_build(CommandFrame *f, uint8_t cmd, uint16_t value)
{
    f->sof      = 0xA5u;
    f->cmd      = cmd;
    f->value    = value;   /* 在 LE 系统上存储 LE*/
    f->checksum = cmd ^ (uint8_t)(value & 0xFFu) ^ (uint8_t)(value >> 8);
    f->eof      = 0x5Au;
}

int frame_validate(const CommandFrame *f)
{
    if (f->sof != 0xA5u || f->eof != 0x5Au) return 0;
    uint8_t expected = f->cmd ^ (uint8_t)(f->value & 0xFFu) ^ (uint8_t)(f->value >> 8);
    return (f->checksum == expected) ? 1 : 0;
}

 * INTERVIEW QUESTION ANSWERS

问题 1：如果从 ISR 标志变量中删除 易失性，会发生什么？

答：如果没有 volatile，编译器会像变量一样分析代码
   永远不会在正常程序流程之外进行更改。在main()中：
     while (!g_flag) {} // 编译器：g_flag 始终为 0 → while(1)

   永远不会重新读取内存。 ISR 写入 RAM，但 main() 读取其

   没有易失性：编译器读取一次并优化以后的读取。
   如果没有 const：编码错误可能会写入 HW 寄存器。

Q3：静态全局与非静态全局。为什么更喜欢静态？

答：非静态全局：通过 `extern` 对 ALL 翻译单元可见。任意.c
   文件可能会被意外读取或修改。
   静态全局（文件范围）：仅在声明 .c 文件中可见。

   - 封装：防止其他模块绕过您的API。
   - 链接器优化：链接器可以消除未使用的静态全局变量。
   - 防止名称冲突：两个文件可以有 `static int counter` 而不会发生冲突。
   嵌入式风格指南 (MISRA C:2012 规则 8.7) 要求变量为 static
   不被其他翻译单位使用。

Q4：SysTick ISR 中的 uint32_t 计数器，在 main 中读取。 M4上的原子？ M0？

   原子性——总线不可分割地完成整个 32 位事务。

   对齐的 32 位访问。

   每次。无论原子性如何，始终使用 `volatile`。
   非原子情况：uint64_t（两个 32 位读取）、结构、未对齐访问。

* TASK 6 — Bug 搜寻 FIXED
 *
 * 错误 1：buf[0] | (buf[1] << 8) — 这是 LITTLE-endian，而不是 big-endian。
 * 对于大端： (buf[0] << 8) |缓冲区[1]。
 * FIX: pkt.temp_centideg = (int16_t)(((uint16_t)buf[0] << 8) | buf[1]);
 *
 * Bug 2: *(uint16_t*)(buf + 2) — 两个问题：
 * (a) 严格别名违规：通过 uint16_t* 访问 uint8_t 数组。
 * (b) 潜在的错位：buf+2 可能不是 2 字节对齐的。
 * 两者都是UB。使用显式的逐字节读取。
 * FIX：原始压力=（uint16_t）（（（uint16_t）buf [2] << 8）| buf [3]）;
 *
 * Bug 3: (buf[4] << 8) — buf[4] 是 uint8_t。当 buf[4] >= 0x80（第 7 位设置）时，
 * 整数提升使其在移位前成为有符号整型，
 * 和符号扩展给出 0xFFFFxx80，使得 OR 结果错误。
 * FIX: ((uint16_t)buf[4] << 8) | buf[5] — 首先转换为 uint16_t。
 * =============================================================== */

typedef struct { int16_t temp_centideg; uint16_t pressure_pa; uint16_t humid_tenth; } SensorPacket;

SensorPacket parse_sensor_packet_FIXED(const uint8_t *buf)
{
    SensorPacket pkt;
    pkt.temp_centideg = (int16_t)(((uint16_t)buf[0] << 8) | buf[1]);   /* 是，已签署 */
    pkt.pressure_pa   = (uint16_t)(((uint16_t)buf[2] << 8) | buf[3]);  /* BE，未签名 */
    pkt.humid_tenth   = (uint16_t)(((uint16_t)buf[4] << 8) | buf[5]);  /* BE，先施放*/
    return pkt;
}

/*===============================================================
 * INTERVIEW QUESTION ANSWERS
 * ===============================================================

Q1：在运行时检测字节序的单行代码。

答：int is_le = (*(uint8_t *)&(uint16_t){1}) == 1;
   说明：复合文字 (uint16_t){1} 存储值 0x0001。
   在 LE 上：存储为 [01][00] → byte[0] == 1 → true。
   BE 上：存储为 [00][01] → byte[0] == 0 → false。

Q2：您通过 TCP 收到 uint32_t。使用之前必须做什么？

A：从网络字节顺序（big-endian）转换为主机字节顺序：
   uint32_t host_val = ntohl(network_val);
   在大端主机上：ntohl() 是无操作。
   在小端主机（x86、ARM）上：ntohl() 反转字节。
   切勿直接使用原始值——这在 LE 主机上会出错。

Q3：CAN 信号 EngineSpeed：start_bit=0，length=16，Intel，factor=0.25，offset=0。
    有效负载[0xE8][0x03]。物理价值？

答：Intel 字节顺序，start_bit=0，length=16。
   raw = 有效负载为 uint64_t LE = 0x03E8 （来自字节 [E8][03] → LE = 0x03E8 = 1000）
   信号 =（原始 >> 0）& 0xFFFF = 1000
   物理 = 1000 * 0.25 + 0 = 250 RPM...等等：
   实际上0x03E8 = 1000。物理= 1000 * 0.25 = 250。
   （但典型的EngineSpeed：因子= 0.25，因此4000 RPM =原始16000 = 0x3E80）

Q4：为什么 *(uint16_t*)(buf+1) 即使在小端 ARM 上也是危险的？

答：有两个原因：
   (1) 严格别名：编译器假定 uint8_t* 和 uint16_t* 从不别名。
       通过 uint16_t* 读取通过 uint8_t* 存储的内容是未定义的行为 -
       编译器可以使用缓存的寄存器值而不是从内存中读取。
   (2) 错位：buf+1 位于奇数地址。在 Cortex-M0/M0+ 上，未对齐
       半字加载会导致 HardFault。在 M3/M4 上它可以工作，但可能会很慢。
   安全的替代方案：memcpy(&val, buf+1, 2);  - 始终定义，始终对齐。

Q5：什么是 htonl()？为什么 UDP 需要它？

答：htonl() = 主机到网络长。从主机字节顺序转换 32 位值
   网络字节顺序（大端）。
   POSIX 套接字期望所有多字节标头字段均按网络字节顺序排列。
   如果您在小端主机上发送不带 htonl() 的 uint32_t：
   值 0x00000001 在内存中存储为 [01][00][00][00]，
   但你需要在线上[00][00][00][01]。
   ntohl() 是相反的：Network TO Host Long（在 recvfrom() 之后使用）。
*/
